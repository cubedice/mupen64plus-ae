cmake_minimum_required(VERSION 3.22.1)
project(slangvulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(RASHADER_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/rashader/${ANDROID_ABI}/liblibrashader_capi.so")
if(NOT EXISTS "${RASHADER_LIBRARY}")
    message(FATAL_ERROR "Build the Vulkan runtime first: bash scripts/build-vulkan-runtime.sh ${ANDROID_ABI}")
endif()
add_library(rashader SHARED IMPORTED)
set_target_properties(rashader PROPERTIES IMPORTED_LOCATION "${RASHADER_LIBRARY}" IMPORTED_NO_SONAME TRUE)
add_library(slangvulkan SHARED slang_vulkan.cpp)
target_compile_definitions(slangvulkan PRIVATE VK_USE_PLATFORM_ANDROID_KHR LIBRA_RUNTIME_VULKAN)
target_link_libraries(slangvulkan rashader vulkan android EGL GLESv2 log)
target_link_options(slangvulkan PRIVATE "-Wl,-z,max-page-size=16384")
